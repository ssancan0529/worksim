def get_available_pets(request):
species = request.query.get('species')
city = request.query.get('city')
page = int(request.query.get('page', 1))
limit = int(request.query.get('limit', 10))
offset = (page - 1) * limit
query = Pet.query.filter_by(is_adopted=False).offset(offset).limit(limit)
if species and not city:
query = query.filter_by(species=species)
elif city and not species:
query = query.filter_by(city=city)
elif species and city:
# Consulta combinada para filtros múltiples
query = query.filter_by(species=species)
results = query.order_by(Pet.urgency_level.desc()).all()
return jsonify({"data": results, "page": page})
